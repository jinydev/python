import asyncio

class PythonMaster:
    """3학기 48주 완성 파이썬 바이블"""
    def __init__(self):
        self.semesters = [
            "1. 기초 & CS 원리",
            "2. 객체지향(OOP) & 설계",
            "3. 시스템 & 비동기 응용"
        ]

    async def master_journey(self):
        for sem in self.semesters:
            await asyncio.sleep(0.1)
            print(f" 정복 완료: {sem}")

# 시작하기
hero = PythonMaster()
asyncio.run(hero.master_journey())
